   1→import express, { Request, Response, NextFunction } from 'express';
   2→import cors from 'cors';
   3→import helmet from 'helmet';
   4→import dotenv from 'dotenv';
   5→import path from 'path';
   6→import { PrismaClient } from '@prisma/client';
   7→import logger from './utils/logger';
   8→import productRoutes from './routes/products';
   9→import newsRoutes from './routes/news';
  10→import formRoutes from './routes/forms';
  11→import chatRoutes from './routes/chat';
  12→import adminRoutes from './routes/admin';
  13→import uploadRoutes from './routes/upload';
  14→import { websocketService } from './utils/websocket';
  15→
  16→// 加载环境变量
  17→dotenv.config();
  18→
  19→const app = express();
  20→const PORT = process.env.PORT || 8080;
  21→
  22→// 强制使用8080端口（Zeabur要求）
  23→const ACTUAL_PORT = 8080;
  24→console.log(`[DEBUG] Environment PORT: ${process.env.PORT}`);
  25→console.log(`[DEBUG] Using port: ${ACTUAL_PORT}`);
  26→
  27→// 初始化 Prisma 客户端
  28→export const prisma = new PrismaClient();
  29→
  30→// 中间件配置
  31→app.use(helmet());
  32→app.use(cors({
  33→  origin: process.env.CORS_ORIGIN || '*',
  34→  credentials: true
  35→}));
  36→app.use(express.json({ limit: '10mb' }));
  37→app.use(express.urlencoded({ extended: true }));
  38→
  39→// 静态文件服务（上传的图片）
  40→app.use('/uploads', express.static(path.resolve(process.cwd(), 'uploads')));
  41→
  42→// 请求日志中间件
  43→app.use((req: Request, res: Response, next) => {
  44→  logger.info(`${req.method} ${req.path}`);
  45→  next();
  46→});
  47→
  48→// 健康检查端点
  49→app.get('/health', (req: Request, res: Response) => {
  50→  res.json({ 
  51→    status: 'ok', 
  52→    timestamp: new Date().toISOString(),
  53→    uptime: process.uptime()
  54→  });
  55→});
  56→
  57→// API 路由
  58→app.use('/api/products', productRoutes);
  59→app.use('/api/news', newsRoutes);
  60→app.use('/api/forms', formRoutes);
  61→app.use('/api/chat', chatRoutes);
  62→app.use('/api/admin', adminRoutes);
  63→app.use('/api/upload', uploadRoutes);
  64→
  65→// 404 处理
  66→app.use((req: Request, res: Response) => {
  67→  res.status(404).json({ error: 'Route not found' });
  68→});
  69→
  70→// 全局错误处理
  71→app.use((err: Error, req: Request, res: Response, next: NextFunction) => {
  72→  logger.error('Unhandled error:', err);
  73→  res.status(500).json({
  74→    error: 'Internal server error',
  75→    message: process.env.NODE_ENV === 'development' ? err.message : undefined
  76→  });
  77→});
  78→
  79→// 启动服务器
  80→const server = app.listen(ACTUAL_PORT, () => {
  81→  logger.info(`Server is running on port ${ACTUAL_PORT}`);
  82→  logger.info(`Environment: ${process.env.NODE_ENV || 'development'}`);
  83→
  84→  // 初始化 WebSocket 服务
  85→  websocketService.initialize(server);
  86→});
  87→
  88→// 优雅关闭
  89→process.on('SIGTERM', async () => {
  90→  logger.info('SIGTERM received. Shutting down gracefully...');
  91→  await prisma.$disconnect();
  92→  server.close(() => {
  93→    logger.info('Server closed');
  94→    process.exit(0);
  95→  });
  96→});
  97→
  98→process.on('SIGINT', async () => {
  99→  logger.info('SIGINT received. Shutting down gracefully...');
 100→  await prisma.$disconnect();
 101→  server.close(() => {
 102→    logger.info('Server closed');
 103→    process.exit(0);
 104→  });
 105→});
 106→
 107→export default app;
