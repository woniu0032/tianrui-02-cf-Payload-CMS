   1→import { Router, Request, Response } from 'express';
   2→import { prisma } from '../index';
   3→import { z } from 'zod';
   4→import logger from '../utils/logger';
   5→
   6→const router = Router();
   7→
   8→// 数据验证 Schema
   9→const productSchema = z.object({
  10→  name: z.string().min(1).max(200),
  11→  description: z.string().max(5000),
  12→  price: z.number().positive(),
  13→  category: z.string().min(1).max(100),
  14→  images: z.array(z.string().url()).optional(),
  15→  content: z.any().optional(), // JSON 富文本内容
  16→  layout: z.any().optional(), // JSON 布局配置
  17→  isActive: z.boolean().optional(),
  18→  sortOrder: z.number().int().optional()
  19→});
  20→
  21→// GET /api/products - 获取产品列表（支持分页和筛选）
  22→router.get('/', async (req: Request, res: Response) => {
  23→  try {
  24→    const { 
  25→      page = 1, 
  26→      limit = 20, 
  27→      category, 
  28→      isActive, 
  29→      search 
  30→    } = req.query;
  31→
  32→    const pageNum = parseInt(page as string);
  33→    const limitNum = parseInt(limit as string);
  34→    const skip = (pageNum - 1) * limitNum;
  35→
  36→    // 构建查询条件
  37→    const where: any = {};
  38→    if (category) where.category = category as string;
  39→    if (isActive !== undefined) where.isActive = isActive === 'true';
  40→    if (search) {
  41→      where.OR = [
  42→        { name: { contains: search as string, mode: 'insensitive' } },
  43→        { description: { contains: search as string, mode: 'insensitive' } }
  44→      ];
  45→    }
  46→
  47→    const [products, total] = await Promise.all([
  48→      prisma.product.findMany({
  49→        where,
  50→        skip,
  51→        take: limitNum,
  52→        orderBy: [{ sortOrder: 'asc' }, { createdAt: 'desc' }]
  53→      }),
  54→      prisma.product.count({ where })
  55→    ]);
  56→
  57→    res.json({
  58→      data: products,
  59→      pagination: {
  60→        page: pageNum,
  61→        limit: limitNum,
  62→        total,
  63→        totalPages: Math.ceil(total / limitNum)
  64→      }
  65→    });
  66→  } catch (error) {
  67→    logger.error('Error fetching products:', error);
  68→    res.status(500).json({ error: 'Failed to fetch products' });
  69→  }
  70→});
  71→
  72→// GET /api/products/:id - 获取单个产品
  73→router.get('/:id', async (req: Request, res: Response) => {
  74→  try {
  75→    const { id } = req.params;
  76→    const product = await prisma.product.findUnique({
  77→      where: { id }
  78→    });
  79→
  80→    if (!product) {
  81→      return res.status(404).json({ error: 'Product not found' });
  82→    }
  83→
  84→    res.json(product);
  85→  } catch (error) {
  86→    logger.error('Error fetching product:', error);
  87→    res.status(500).json({ error: 'Failed to fetch product' });
  88→  }
  89→});
  90→
  91→// POST /api/products - 创建产品
  92→router.post('/', async (req: Request, res: Response) => {
  93→  try {
  94→    const validatedData = productSchema.parse(req.body);
  95→    
  96→    const product = await prisma.product.create({
  97→      data: validatedData
  98→    });
  99→
 100→    logger.info(`Product created: ${product.id}`);
 101→    res.status(201).json(product);
 102→  } catch (error) {
 103→    if (error instanceof z.ZodError) {
 104→      return res.status(400).json({ 
 105→        error: 'Validation failed', 
 106→        details: error.errors 
 107→      });
 108→    }
 109→    logger.error('Error creating product:', error);
 110→    res.status(500).json({ error: 'Failed to create product' });
 111→  }
 112→});
 113→
 114→// PUT /api/products/:id - 更新产品
 115→router.put('/:id', async (req: Request, res: Response) => {
 116→  try {
 117→    const { id } = req.params;
 118→    const validatedData = productSchema.partial().parse(req.body);
 119→
 120→    const product = await prisma.product.update({
 121→      where: { id },
 122→      data: validatedData
 123→    });
 124→
 125→    logger.info(`Product updated: ${product.id}`);
 126→    res.json(product);
 127→  } catch (error) {
 128→    if (error instanceof z.ZodError) {
 129→      return res.status(400).json({ 
 130→        error: 'Validation failed', 
 131→        details: error.errors 
 132→      });
 133→    }
 134→    logger.error('Error updating product:', error);
 135→    res.status(500).json({ error: 'Failed to update product' });
 136→  }
 137→});
 138→
 139→// DELETE /api/products/:id - 删除产品
 140→router.delete('/:id', async (req: Request, res: Response) => {
 141→  try {
 142→    const { id } = req.params;
 143→    
 144→    await prisma.product.delete({
 145→      where: { id }
 146→    });
 147→
 148→    logger.info(`Product deleted: ${id}`);
 149→    res.status(204).send();
 150→  } catch (error) {
 151→    logger.error('Error deleting product:', error);
 152→    res.status(500).json({ error: 'Failed to delete product' });
 153→  }
 154→});
 155→
 156→// PATCH /api/products/:id/layout - 更新产品布局（拖拽编辑器专用）
 157→router.patch('/:id/layout', async (req: Request, res: Response) => {
 158→  try {
 159→    const { id } = req.params;
 160→    const { layout, content } = req.body;
 161→
 162→    const product = await prisma.product.update({
 163→      where: { id },
 164→      data: {
 165→        ...(layout && { layout }),
 166→        ...(content && { content })
 167→      }
 168→    });
 169→
 170→    logger.info(`Product layout updated: ${product.id}`);
 171→    res.json(product);
 172→  } catch (error) {
 173→    logger.error('Error updating product layout:', error);
 174→    res.status(500).json({ error: 'Failed to update layout' });
 175→  }
 176→});
 177→
 178→export default router;
